def on_append_entries(self, term, leader_id, prev_log_index, prev_log_term, entries, leader_commit):
    # ① termが古ければ拒否
    if term < self.current_term:
        self.append_entries_success = False
        return

    self.current_term = term

    # ② prevLogIndex のチェック（-1 は初期状態なのでスキップ）
    if prev_log_index >= 0:
        # ログが足りない場合
        if prev_log_index >= len(self.log):
            self.append_entries_success = False
            return

        # タームが一致しない場合（衝突）
        if self.log[prev_log_index]["term"] != prev_log_term:
            # 衝突部分を削除
            self.log = self.log[:prev_log_index]
            self.append_entries_success = False
            return

    # ③ エントリを追加（衝突チェックしながら）
    index = prev_log_index + 1
    for i, entry in enumerate(entries):
        log_index = index + i
        if log_index < len(self.log):
            # 既存のエントリと新しいエントリのタームを比較
            if self.log[log_index]["term"] != entry["term"]:
                # 衝突 → ここから削除して新しいエントリを追加
                self.log = self.log[:log_index]
                self.log.extend(entries[i:])
                break
        else:
            # 既存のログの範囲外 → 新しいエントリを追加
            self.log.extend(entries[i:])
            break

    # ④ commitIndex 更新
    if leader_commit > self.commit_index:
        self.commit_index = min(leader_commit, len(self.log) - 1)
        self.state_machine[self.id] = self.commit_index

    self.append_entries_success = True
